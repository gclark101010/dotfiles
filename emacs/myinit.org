** Poetry
#+begin_src emacs-lisp
(use-package poetry
:ensure t)

;; Enable Poetry tracking mode globally
(poetry-tracking-mode 1)

;; Set the strategy to use Projectile for switching environments
(setq poetry-tracking-strategy 'projectile)

;; Guard against poetry-tracking-mode activating invalid paths as virtualenvs.
;; When a project has no poetry venv configured, `poetry env info -p` returns
;; nothing; poetry.el then calls (pyvenv-activate "") which expands "" to
;; default-directory (the source file's directory). jediepcserver.py reads
;; VIRTUAL_ENV and tries to use that directory as a Python env, crashing.
(defun my/pyvenv-activate-guard (orig-fun directory &rest args)
  "Skip activation if DIRECTORY is not a real virtualenv."
  (let ((python (expand-file-name "bin/python" directory)))
    (if (file-executable-p python)
        (apply orig-fun directory args)
      nil)))

(advice-add 'pyvenv-activate :around #'my/pyvenv-activate-guard)

;; Suppress the *poetry-error* popup when the buffer is empty.
;; This happens when `poetry env info -p` exits non-zero because no virtualenv
;; is configured — the command produces no output, leaving an empty buffer that
;; still gets displayed. Real errors have content and are unaffected.
(defun my/poetry-suppress-empty-error-buffer (&rest _)
  "Close *poetry-error* if it has no content (noise from unconfigured venv)."
  (when-let ((buf (get-buffer (poetry-buffer-name "error"))))
    (when (string-blank-p (with-current-buffer buf (buffer-string)))
      (delete-windows-on buf)
      (kill-buffer buf))))

(advice-add 'poetry--run-next-call-from-queue :after #'my/poetry-suppress-empty-error-buffer)
#+end_src

* Pass Generator
source: [[https://www.emacswiki.org/emacs/PasswordGenerator][EmacsWiki: Password Generator]]

#+begin_src emacs-lisp
  (cl-defun make-password (length &optional (upper t) (lower t) (number t) (symbol nil) (ambiguous nil))
    "Return a string of LENGTH random characters.  If UPPER is non-nil,
  use uppercase letters.  If lower is non-nil, use lowercase letters.
  If NUMBER is non-nil, use numbers.  If SYMBOL is non-nil, use one of
  \"!\"#$%&'()*+'-./:;<=>?@`{}|~\".  If AMBIGUOUS is nil, avoid
  characters like \"l\" and \"1\", \"O\" and \"0\"."
    (interactive (make-password-prompt-for-args))
    (let ((char-list (make-password-char-list upper lower number symbol ambiguous))
           position password)
      (random t)
    (cl-loop for i from 1 to length
          do (setq position (random (length char-list))
                   password (concat password (string (nth position char-list)))))
    (if (called-interactively-p 'any)
        (let* ((strength (make-password-strength length upper lower number symbol ambiguous))
               (bits (car strength))
               (number (cadr strength)))
          (message "The password \"%s\" is one of 10^%d possible and has a bit equivalence of %d"
                   password (round number) (round bits)))
      password)))

  (defun make-password-char-list (upper lower number symbol ambiguous)
    (let* ((upper-chars-ambiguous '(?I ?O ?G))
           (upper-chars (cl-loop for i from ?A to ?Z unless
                              (member i upper-chars-ambiguous)
                              collect i))
           (lower-chars-ambiguous '(?l ?o))
           (lower-chars (cl-loop for i from ?a to ?z unless
                              (member i lower-chars-ambiguous)
                              collect i))
           (number-chars-ambiguous '(?0 ?1 ?6))
           (number-chars (cl-loop for i from ?0 to ?9 unless
                               (member i number-chars-ambiguous)
                               collect i))
           (symbol-chars '(?! ?@ ?# ?$ ?% ?& ?* ?( ?) ?+ ?= ?/
                              ?{ ?} ?[ ?] ?: ?\; ?< ?>))
           (symbol-chars-ambiguous '(?_ ?- ?| ?, ?. ?` ?' ?~ ?^ ?\"))
           char-list)
    (if upper
        (setq char-list (append char-list upper-chars)))
    (if lower
        (setq char-list (append char-list lower-chars)))
    (if number
        (setq char-list (append char-list number-chars)))
    (if symbol
        (setq char-list (append char-list symbol-chars)))
    (if ambiguous
        (setq char-list (append char-list
                                upper-chars-ambiguous
                                lower-chars-ambiguous
                                number-chars-ambiguous
                                symbol-chars-ambiguous)))
    char-list))

  (defun make-password-prompt-for-args ()
    (interactive)
    (list
     (string-to-number (read-from-minibuffer "Number of Characters: "))
     (y-or-n-p "User uppercase: ")
     (y-or-n-p "User lowercase: ")
     (y-or-n-p "User numbers: ")
     (y-or-n-p "User symbols: ")
     (y-or-n-p "User ambiguous characters: ")))

  (cl-defun make-password-strength (length &optional (upper t) (lower t) (number t) (symbol nil) (ambiguous nil))
    "Calculate the number of possible passwords that could be generated
  given the criteria of LENGTH and use of UPPER, LOWER, NUMBER, SYMBOL,
  and AMBIGUOUS characters"
    (interactive (make-password-prompt-for-args))
    (let* ((char-list (make-password-char-list upper lower number symbol ambiguous))
           (bits (/ (* length (log (length char-list))) (log 2)))
           (number (/ (* bits (log 2)) (log 10))))
      (if (called-interactively-p 'any)
          (message "number of combinations is 10^%d with a bit equivalence of %d" (round number) (round bits))
        (list bits number))))
